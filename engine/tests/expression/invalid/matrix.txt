# Invalid matrix expressions — each must throw a UserError.

# Ragged rows / empty literal (unrecognised input)
[1 2, 3]
[]
[1 2, 3 4,]

# Shape mismatches
[1 2, 3 4] + [1 2 3, 4 5 6]
[1 2, 3 4] * [1 2 3]
hadamard([1 2, 3 4], [1 2 3, 4 5 6])

# Non-square where a square matrix is required
determinant([1 2 3, 4 5 6])
inverse([1 2 3, 4 5 6])
trace([1 2 3, 4 5 6])

# Singular matrix has no inverse
inverse([1 2, 2 4])

# Power must be an integer; a matrix exponent is undefined
[1 2, 3 4] ^ 1.5
[1 2, 3 4] ^ [1 0, 0 1]

# scalar / matrix is undefined
5 / [1 2, 3 4]

# A matrix function needs a matrix argument
transpose(5)
determinant(7)

# Complex eigenvectors are not supported
eigenvectors([0 -1, 1 0])
